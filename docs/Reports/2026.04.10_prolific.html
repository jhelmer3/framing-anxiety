The document used data that I think was not recoded to align with scale direction. I renamed the values on Qualtrics, adding a _r tag to variables that need recoding. When I create recoded variables below, they will not have the _r tag. I also changed the BIS / BAS variable names to point specifically to their subscale (BIS or BAS).

Decisions

Code 
dat <- dat_full |>
  select(id, matches("^[F][GL][PV]_[0-9\\.-]+$")) |>
  pivot_longer(!id,
               names_to = "item", values_to = "response") |>
  filter(!is.na(response)) |>
  separate_wider_regex(item, c(cond = "[F][GL][PV]", "_", rest = ".*")) |>
  separate_wider_regex(rest, c(fixed = "^\\d*", "-*", varied = ".*")) |>
  separate_wider_position(cond, c(1, "frame" = 1, "target" = 1)) |>
  mutate(.by = c(frame, target, fixed),
         item = cur_group_id(),
         .after = id) |>
  mutate(.by = c(item, id),
         trial = row_number(),
         .after = id) |>
  mutate(varied = as.numeric(varied),
         fixed = as.numeric(fixed),
         lottery = ifelse(target == "P", varied, fixed),
         certain = ifelse(target == "V", varied, fixed),
         ev_lottery = ifelse(frame == "L", 100 - lottery, lottery),
         ev_certain = ifelse(frame == "L", 100 - certain, certain),
         
         lottery_std = (lottery - mean(lottery)) / sd(lottery),
         certain_std = (certain - mean(certain)) / sd(lottery),
         
         # if matches the fixed dollar option, 0, else 1 (for risky)
         chose_risky = ifelse(str_detect(response, "Keep \\$[0-9]+") | 
                                str_detect(response, "Forfeit \\$[0-9]+"), 0, 1)) |>
  left_join(IUS |>
              pivot_longer(!id, names_to = "item", values_to = "response") |>
              summarize(.by = id,
                        IUS = mean(response)),
            by = "id") |>
  left_join(BIS |>
              pivot_longer(!id, names_to = "item", values_to = "response") |>
              summarize(.by = id,
                        BIS = mean(response)),
            by = "id") |>
  left_join(BAS |>
              pivot_longer(!id, names_to = "item", values_to = "response") |>
              summarize(.by = id,
                        BAS = mean(response)),
            by = "id") |>
  left_join(STAI |>
              pivot_longer(!id, names_to = "item", values_to = "response") |>
              summarize(.by = id,
                        STAI_T = mean(response)),
            by = "id") |>
  mutate(lottery_certain_diff = ev_lottery - ev_certain,
         across(c(IUS, BIS, BAS, STAI_T), ~ (.x - mean(.x)) / sd(.x),
                .names = "{.col}_z"))
